"""Regenerate allocation figures from the checksum-verified, nine-decimal extract.
Each output has one axes. Bars show the four-parent range, not confidence intervals.
No training, checkpoint loading, or network access.
"""
from pathlib import Path
import json, hashlib
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

ROOT = Path(__file__).resolve().parents[1]
data_path = ROOT / "evidence/budget_interpretability/plot_data_compact.json"
assert hashlib.sha256(data_path.read_bytes()).hexdigest() == \
    "996a8bf80aa746c9aa53d9cf359e244f0b6c8fa2c2efdd93ab84a0930daea93b"
data = json.loads(data_path.read_text())
v = np.asarray(data["values"], dtype=float)
assert v.shape == (4, 2, 10, 3, 3)
x = np.asarray(data["phases"])
rho_s = 2.0 ** np.clip((x - 200) / 200, 0, 1)
out = ROOT / "figures/allocation"
out.mkdir(parents=True, exist_ok=True)

def range_curve(ax, values, label, marker, linestyle):
    mean = values.mean(axis=0)
    spread = np.stack((mean-values.min(axis=0), values.max(axis=0)-mean))
    ax.errorbar(x, mean, yerr=spread, label=label, marker=marker,
                linestyle=linestyle, markersize=3.0, linewidth=1.1,
                elinewidth=0.75, capsize=1.4)

def finish(fig, ax, filename, ylabel, ylim, yticks, title, legend=True):
    ax.set_title(title, fontsize=9, pad=4)
    ax.set_xlabel("Post-growth update", fontsize=8, labelpad=2)
    ax.set_ylabel(ylabel, fontsize=8, labelpad=2)
    ax.set_xlim(100, 3300)
    ax.set_xticks([400, 1500, 3000])
    ax.set_ylim(*ylim)
    ax.set_yticks(yticks)
    ax.tick_params(axis="both", labelsize=7.8, length=2.5, pad=2)
    ax.spines["top"].set_visible(False)
    ax.spines["right"].set_visible(False)
    if legend:
        handles, labels = ax.get_legend_handles_labels()
        order = sorted(range(len(labels)), key=lambda i: not labels[i].startswith("Full"))
        ax.legend([handles[i] for i in order], [labels[i] for i in order],
                  fontsize=7.5, frameon=False, loc="lower right",
                  handlelength=1.7, borderpad=.15, labelspacing=.25)
    fig.subplots_adjust(left=.25, bottom=.23, right=.98, top=.88)
    fig.savefig(out / (filename+".pdf"))
    fig.savefig(out / (filename+".png"), dpi=180)
    plt.close(fig)

for bi, branch in enumerate(data["branches"]):
    stem = "full_state" if bi == 0 else "structured_state"
    for pi, proj in enumerate(data["projections"]):
        qf, qs = v[:,bi,:,pi,0], v[:,bi,:,pi,1]
        qr = qs/(rho_s[None,:]**2*(1-qs)+qs)
        reff = rho_s[None,:]*np.sqrt(qf*(1-qs)/((1-qf)*qs))
        title = {"gate":"Gate projection","up":"Up projection","down":"Down projection"}[proj]

        fig, ax = plt.subplots(figsize=(1.85,1.55))
        range_curve(ax,qf,"Full BDT","o","-")
        range_curve(ax,qs,"Structured","s","--")
        finish(fig,ax,f"q_{stem}_{proj}",
               r"Fraction $q_D$",(.055,.55),[.1,.3,.5],title)

        fig, ax = plt.subplots(figsize=(1.85,1.55))
        range_curve(ax,reff,"Full BDT","o","-")
        ax.plot(x,rho_s,label="Structured",marker="s",
                linestyle="--",markersize=3,linewidth=1.1)
        finish(fig,ax,f"rho_{stem}_{proj}",
               r"Effective scale $\rho_{\rm eff}$",(.9,2.45),[1,1.5,2,2.4],title)

print("Saved 12 allocation figures; each PDF/PNG contains one projection and one axes.")
